	' MiniHtml Boilerplate class
	' Version: 4.00
	Private mContent As MiniHtml
End Sub

Public Sub Initialize

End Sub

Public Sub LoadContent (Tag1 As MiniHtml)
	mContent = Tag1
End Sub

Public Sub Render As MiniHtml
	Dim page1 As MiniHtml = MH.Html
	Dim head1 As MiniHtml = MH.ResponsiveHeader.up(page1)
	MH.Title.up(head1).text(Main.App.ctx.Get("APP_TITLE"))
	MH.Link("icon", "image/png").up(head1).href(Main.App.ServerUrl & "/assets/img/favicon.png")
	MH.Link.up(head1).href(Main.App.ServerUrl & "/assets/css/bootstrap.min.css")
	Dim body1 As MiniHtml = MH.Body.up(page1)
	MH.H1.up(body1).text("MiniHtml")
	If Initialized(mContent) Then mContent.up(body1)
	MH.Script.up(body1).src(Main.App.ServerUrl & "/assets/js/bootstrap.min.js")
	Return page1
